/**
 * ============================================================================
 * QUESTION: Path Sum II (LeetCode 113)
 * ============================================================================
 * Given the root of a binary tree and an integer targetSum, return all 
 * root-to-leaf paths where the sum of the node values in the path equals 
 * targetSum. Each path should be returned as a list of the node values, 
 * not node references.
 *
 * ============================================================================
 * APPROACH: Backtracking with DFS
 * ============================================================================
 * 1. Tracking State:
 *    - Maintain a `currentPath` list to accumulate nodes as we walk down the tree.
 *    - Subtract each visited node's value from `remaining`.
 *
 * 2. Leaf Node & Target Validation:
 *    - When we reach a leaf node (`node.left == null && node.right == null`) 
 *      AND `remaining == 0`, we found a valid path.
 *    - Crucial Step: Add a *new copy* (`new ArrayList<>(currentPath)`) to the 
 *      results list because `currentPath` will continue mutating during backtracking.
 *
 * 3. Backtracking:
 *    - After exploring both left and right subtrees, remove the last node from 
 *      `currentPath` (`currentPath.remove(...)`) before returning to the parent. 
 *      This resets the path state for alternative branches.
 *
 * 4. Complexity:
 *    - Time:  O(N^2) in the worst case (where N is the number of nodes), 
 *             because copying paths of length up to N takes O(N) time per leaf.
 *    - Space: O(N) to store the recursion stack and path states.
 * ============================================================================
 */

import java.util.*;

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<List<Integer>> pathSum(TreeNode root, int targetSum) {
        List<List<Integer>> result = new ArrayList<>();
        List<Integer> currentPath = new ArrayList<>();
        dfs(root, targetSum, currentPath, result);
        return result;
    }

    private void dfs(TreeNode node, int remaining, List<Integer> currentPath, List<List<Integer>> result) {
        // 1. Base case: if node is null, stop recursion
        if (node == null) return;

        // 2. Choose: add current node to the path and reduce the remaining target
        currentPath.add(node.val);
        remaining -= node.val;

        // 3. Check if we reached a valid leaf node with a matching sum
        if (node.left == null && node.right == null && remaining == 0) {
            result.add(new ArrayList<>(currentPath)); // Copy list to avoid reference mutation bugs
        } else {
            // Otherwise, explore left and right subtrees
            dfs(node.left, remaining, currentPath, result);
            dfs(node.right, remaining, currentPath, result);
        }

        // 4. Un-choose (Backtrack): remove the last node before going back up to the parent
        currentPath.remove(currentPath.size() - 1);
    }
}
