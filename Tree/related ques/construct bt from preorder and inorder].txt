/**
 * ============================================================================
 * QUESTION: Construct Binary Tree from Preorder and Inorder Traversal (LeetCode 105)
 * ============================================================================
 * Given two integer arrays `preorder` and `inorder` where `preorder` is the 
 * preorder traversal of a binary tree and `inorder` is the inorder traversal 
 * of the same tree, construct and return the binary tree.
 *
 * ============================================================================
 * APPROACH: Divide & Conquer using HashMap
 * ============================================================================
 * 1. Key Traversal Properties:
 *    - Preorder (Root -> Left -> Right):
 *        The first element is always the current subtree's root.
 *    - Inorder (Left -> Root -> Right):
 *        Elements to the left of the root belong to the left subtree, and 
 *        elements to the right belong to the right subtree.
 *
 * 2. Fast Lookups with HashMap:
 *    - Store (value -> index) pairs for `inorder` in a HashMap to locate any 
 *      root's position (`mid`) in O(1) time.
 *
 * 3. Calculating Subtree Offsets in Preorder:
 *    - Left subtree size = `mid - left`.
 *    - Left subtree root in preorder = `rootindx + 1`.
 *    - Right subtree root in preorder = `rootindx + (left subtree size) + 1` 
 *                                    = `rootindx + mid - left + 1`.
 *
 * 4. Complexity:
 *    - Time:  O(N) -> Each node is constructed once, and HashMap lookups are O(1).
 *    - Space: O(N) -> O(N) for HashMap storage + O(H) recursion stack space.
 * ============================================================================
 */

import java.util.*;

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public TreeNode buildTree(int[] preorder, int[] inorder) {
        // Precompute indices of inorder elements for O(1) boundary lookups
        Map<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < inorder.length; i++) {
            map.put(inorder[i], i);
        }

        // Recursively split and build tree across the full inorder range [0, inorder.length - 1]
        return split(preorder, map, 0, 0, inorder.length - 1);
    }

    private TreeNode split(int[] preorder, Map<Integer, Integer> map, int rootindx, int left, int right) {
        // 1. Construct the current root node using the preorder index
        TreeNode root = new TreeNode(preorder[rootindx]);

        // 2. Find root position in the inorder array
        int mid = map.get(preorder[rootindx]);

        // 3. Build left subtree if left elements exist (inorder range: left to mid - 1)
        if (mid > left) {
            root.left = split(preorder, map, rootindx + 1, left, mid - 1);
        }

        // 4. Build right subtree if right elements exist (inorder range: mid + 1 to right)
        // Root index in preorder skips all left subtree nodes: rootindx + (mid - left) + 1
        if (mid < right) {
            root.right = split(preorder, map, rootindx + mid - left + 1, mid + 1, right);
        }

        return root;
    }
}
