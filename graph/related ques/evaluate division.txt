import java.util.*;

class Solution {
    /**
     * Problem: LeetCode 399 - Evaluate Division
     *
     * Approach: Graph Modeling + Depth-First Search (DFS)
     * - Graph Representation: Map<Node, Map<Neighbor, Weight>>
     *   - Edge u -> v with weight W represents (u / v = W).
     *   - Edge v -> u with weight 1/W represents (v / u = 1 / W).
     * - Path Evaluation:
     *   - Evaluating C / D amounts to finding any path from C to D and accumulating 
     *     the product of weights along the path.
     *   - A Set<String> visited tracks visited nodes per query to prevent cycles.
     *
     * Complexity Analysis:
     * - Let N be the number of input equations and Q be the number of queries.
     * - Let V be the number of unique variables (V <= 2N) and E = 2N edges.
     * - Time Complexity:
     *   - Graph construction: O(N)
     *   - Query evaluation: O(Q * (V + E)) in the worst case per query.
     * - Space Complexity:
     *   - Graph storage: O(V + E)
     *   - Visited set & recursion stack per query: O(V)
     */
    public double[] calcEquation(List<List<String>> equations, double[] values, List<List<String>> queries) {
        // Step 1: Build the directed, weighted adjacency graph.
        Map<String, Map<String, Double>> graph = new HashMap<>();

        for (int i = 0; i < equations.size(); i++) {
            String u = equations.get(i).get(0);
            String v = equations.get(i).get(1);
            double val = values[i];

            graph.putIfAbsent(u, new HashMap<>());
            graph.putIfAbsent(v, new HashMap<>());

            // u / v = val  ==>  u -> v with weight val
            graph.get(u).put(v, val);
            // v / u = 1.0 / val  ==>  v -> u with weight (1.0 / val)
            graph.get(v).put(u, 1.0 / val);
        }

        // Step 2: Answer each query by traversing the graph.
        double[] results = new double[queries.size()];

        for (int i = 0; i < queries.size(); i++) {
            String start = queries.get(i).get(0);
            String target = queries.get(i).get(1);

            // Case 1: Either node was never introduced in any equation.
            if (!graph.containsKey(start) || !graph.containsKey(target)) {
                results[i] = -1.0;
            } 
            // Case 2: Querying identical known variables (e.g., a / a = 1.0).
            else if (start.equals(target)) {
                results[i] = 1.0;
            } 
            // Case 3: Search for an indirect or direct multiplicative path.
            else {
                Set<String> visited = new HashSet<>();
                results[i] = dfs(graph, start, target, visited, 1.0);
            }
        }

        return results;
    }

    /**
     * Traverses the graph to find a path from curr to target.
     *
     * @param graph        Adjacency list with weighted directed edges.
     * @param curr         Current node being expanded.
     * @param target       Destination variable node.
     * @param visited      Set of already visited variables in the current query path.
     * @param accProduct   Cumulative product of edge weights along the current path.
     * @return Cumulative quotient if a path is found, otherwise -1.0.
     */
    private double dfs(Map<String, Map<String, Double>> graph, String curr, String target,
                        Set<String> visited, double accProduct) {
        // Target reached: return the accumulated ratio product.
        if (curr.equals(target)) {
            return accProduct;
        }

        visited.add(curr);

        // Traverse all adjacent variable relations.
        for (Map.Entry<String, Double> neighbor : graph.get(curr).entrySet()) {
            String nextNode = neighbor.getKey();
            double weight = neighbor.getValue();

            if (!visited.contains(nextNode)) {
                double result = dfs(graph, nextNode, target, visited, accProduct * weight);
                // If a valid path is found down this branch, propagate the value upward.
                if (result != -1.0) {
                    return result;
                }
            }
        }

        // Target cannot be reached via this branch.
        return -1.0;
    }
}
