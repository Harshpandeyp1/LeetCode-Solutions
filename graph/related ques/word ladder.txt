import java.util.*;

class Solution {
    /**
     * Problem: LeetCode 127 - Word Ladder
     *
     * Approach: Breadth-First Search (BFS) / Level-Order Traversal
     * - Each valid word is a node in an unweighted graph.
     * - Two words have an edge if they differ by exactly 1 character.
     * - BFS guarantees that the first time we encounter `endWord`, we have found
     *   the shortest transformation sequence.
     *
     * In-Place Character Mutation:
     * - For each word of length L, generate all possible 26 * L single-character variations.
     * - Modifying a char[] array in-place and checking a HashSet is significantly faster
     *   than using substring() operations or checking all pairs in wordList.
     * - Once a word is placed in the queue, remove it from the Set to mark it as visited.
     *
     * Complexity Analysis:
     * - Let N = number of words in wordList, L = length of each word.
     * - Time Complexity:  O(N * L * 26) = O(N * L) since at most N words are visited,
     *                     and each generates 26 * L candidate strings.
     * - Space Complexity: O(N * L) to store the dictionary Set and the BFS Queue.
     */
    public int ladderLength(String beginWord, String endWord, List<String> wordList) {
        // Step 1: Put all words in a HashSet for O(1) existence lookup and visited tracking.
        Set<String> wordSet = new HashSet<>(wordList);

        // If endWord is not in the dictionary, no valid sequence can be formed.
        if (!wordSet.contains(endWord)) {
            return 0;
        }

        // Step 2: Initialize BFS queue with the beginWord.
        Queue<String> queue = new LinkedList<>();
        queue.offer(beginWord);

        // Sequence length starts at 1 (the beginWord itself counts as step 1).
        int level = 1;

        // Step 3: Perform level-order BFS traversal.
        while (!queue.isEmpty()) {
            int size = queue.size();

            // Process all words at the current transformation depth.
            for (int i = 0; i < size; i++) {
                String word = queue.poll();

                // Target reached: return the current transformation length.
                if (word.equals(endWord)) {
                    return level;
                }

                // Step 4: Generate all adjacent 1-letter mutation neighbors.
                char[] chars = word.toCharArray();
                for (int j = 0; j < chars.length; j++) {
                    char originalChar = chars[j];

                    for (char c = 'a'; c <= 'z'; c++) {
                        if (c == originalChar) {
                            continue;
                        }

                        chars[j] = c;
                        String nextWord = new String(chars);

                        // If valid neighbor exists in dictionary, add to queue and mark visited
                        if (wordSet.contains(nextWord)) {
                            queue.offer(nextWord);
                            wordSet.remove(nextWord); // Remove to prevent cyclic/duplicate visits
                        }
                    }

                    // Backtrack character array to explore the next position
                    chars[j] = originalChar;
                }
            }

            // Move to the next sequence level
            level++;
        }

        // No valid transformation sequence exists to reach endWord.
        return 0;
    }
}
