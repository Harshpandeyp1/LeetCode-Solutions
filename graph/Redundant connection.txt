class Solution {
    Map<Integer, List<Integer>> graph = new HashMap<>();

    public int[] findRedundantConnection(int[][] edges) {
        for (int[] edge : edges) {
            int u = edge[0], v = edge[1];

            if (hasPath(u, v, new HashSet<>())) {
                return edge;
            }

            graph.computeIfAbsent(u, k -> new ArrayList<>()).add(v);
            graph.computeIfAbsent(v, k -> new ArrayList<>()).add(u);
        }
        return new int[0];
    }

    private boolean hasPath(int curr, int target, Set<Integer> visited) {
        if (curr == target) return true;
        visited.add(curr);

        for (int neighbor : graph.getOrDefault(curr, new ArrayList<>())) {
            if (!visited.contains(neighbor) && hasPath(neighbor, target, visited)) {
                return true;
            }
        }
        return false;
    }
}
