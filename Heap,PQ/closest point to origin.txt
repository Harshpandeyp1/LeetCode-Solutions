Given an array of points where points[i] = [xi, yi] represents a point on the X-Y plane and an integer k, return the k closest points to the origin (0, 0).

The distance between two points on the X-Y plane is the Euclidean distance (i.e., √(x1 - x2)2 + (y1 - y2)2).

You may return the answer in any order. The answer is guaranteed to be unique (except for the order that it is in).

 

import java.util.PriorityQueue;

class Solution {
    public int[][] kClosest(int[][] points, int k) {
        
        // Create a max-heap of points, ordered by squared distance from origin
        // The point with the LARGEST distance sits at the top of this heap
        PriorityQueue<int[]> pq = new PriorityQueue<>(
            (a, b) -> (b[0]*b[0] + b[1]*b[1]) - (a[0]*a[0] + a[1]*a[1])
        );
        
        // Go through every point one at a time
        for (int[] p : points) {
            
            // Let this point into the "club" (heap), no matter what
            pq.offer(p);
            
            // If club now has more than k members, kick out the farthest one
            // (the top of the max-heap is always the current farthest point)
            if (pq.size() > k) {
                pq.poll();
            }
        }
        
        // At this point, the heap has exactly k points left —
        // these are the k closest points to the origin
        int[][] res = new int[k][2];
        for (int i = 0; i < k; i++) {
            res[i] = pq.poll(); // take each remaining point out one by one
        }
        
        return res;
    }
}
