package leetcode.medium;

import java.util.Stack;

/*
 * QUESTION: Decode String (LeetCode 394)
 * ---------------------------------------------------
 * Given an encoded string, return its decoded string.
 * Encoding rule: k[encoded_string] means the encoded_string inside the
 * brackets is repeated exactly k times.
 *
 * Example:
 *   "3[a2[c]]" -> "accaccacc"
 *
 * APPROACH: Single Stack with "[" as a boundary marker
 * ---------------------------------------------------
 * Instead of maintaining a separate "current StringBuilder" that gets reset
 * and restored (as in the two-StringBuilder-stack approach), this version
 * pushes EVERY token — digits' counts (on their own stack), individual
 * characters, and even the "[" bracket itself — onto one stringStack.
 *
 * When a ']' is hit, we know the matching "[" is somewhere below in the
 * stack. So we pop everything above it (that's the inner decoded segment),
 * reverse-build it into `temp`, discard the "[", pop the count that goes
 * with it, repeat `temp` that many times, and push the result back onto
 * the stack as if it were a single unit — ready to be used by an outer
 * bracket later (handles nesting naturally).
 *
 * Steps:
 *   1. Traverse each character.
 *   2. Digit -> accumulate into k (handles multi-digit numbers).
 *   3. '[' -> push count k onto numStack, reset k, push "[" onto stringStack
 *      as a marker.
 *   4. Any other letter (not ']') -> push it as a single-char string onto
 *      stringStack.
 *   5. ']' -> pop from stringStack until we hit "[", accumulating into
 *      `temp` (inserting at front to preserve order since we're popping
 *      in reverse). Discard the "[". Pop the matching count from numStack.
 *      Repeat `temp` count times, push the resulting string back onto
 *      stringStack (this becomes one "token" that an outer bracket, if
 *      any, will later scoop up along with everything else).
 *   6. At the end, concatenate everything left in stringStack (bottom to
 *      top) to get the final decoded string.
 *
 * Time Complexity: O(n * maxRepeat) in the worst case of heavy nesting,
 *                   generally O(n) for the final decoded output length.
 * Space Complexity: O(n) for the two stacks.
 */
public class DecodeString {
  public String decodeString(String s) {
    Stack<Integer> numStack = new Stack<>();
    Stack<String> stringStack = new Stack<>();
    int k = 0;

    for (char c : s.toCharArray()) {
      if (Character.isDigit(c)) {
        // Build multi-digit numbers, e.g. "12[a]" -> k = 12
        k = (k * 10) + (c - '0');
        continue;
      }

      if (c == '[') {
        // Save the repeat count, reset k for any nested number,
        // and drop a "[" marker so we know where this segment starts
        numStack.push(k);
        k = 0;
        stringStack.push(String.valueOf(c));
        continue;
      }

      if (c != ']') {
        // Regular letter — push as its own string
        stringStack.push(String.valueOf(c));
        continue;
      }

      // c == ']' : time to resolve the innermost complete bracket segment
      StringBuilder temp = new StringBuilder();

      // Pop everything until we hit the matching "[" marker,
      // inserting at the front since pop() gives us reverse order
      while (!stringStack.peek().equals("[")) {
        temp.insert(0, stringStack.pop());
      }

      // Discard the "[" marker itself
      stringStack.pop();

      // Repeat the segment `count` times
      StringBuilder replacement = new StringBuilder();
      int count = numStack.pop();
      for (int i = 0; i < count; i++) {
        replacement.append(temp);
      }

      // Push the fully expanded segment back as one unit,
      // so an outer "[" (if this was nested) can pick it up later
      stringStack.push(replacement.toString());
    }

    // Concatenate whatever remains on the stack (bottom to top)
    StringBuilder result = new StringBuilder();
    while (!stringStack.empty()) {
      result.insert(0, stringStack.pop());
    }
    return result.toString();
  }
}
