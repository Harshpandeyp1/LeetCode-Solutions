/*
 * QUESTION:
 * You are given an integer array `cardPoints`, where cardPoints[i] represents 
 * the points of the i-th card. You are also given an integer k.
 * In one step, you can take one card from the beginning or from the end of 
 * the array. You must take exactly k cards in total.
 * Your score is the sum of the points of the cards you took.
 * Return the maximum score you can get.
 *
 * APPROACH:
 * - We must pick exactly k cards, and only from the front or the back.
 * - This means the cards we do NOT pick will always form one single 
 *   continuous (contiguous) block somewhere in the middle of the array.
 * - So instead of trying all combinations of front/back picks, we flip 
 *   the problem: find the minimum sum of a contiguous block of size 
 *   (n - k) — because whatever is left after removing that block is 
 *   exactly the cards we picked.
 * - We use a sliding window of size (n - k) to find this minimum sum 
 *   efficiently in one pass.
 * - Final answer = totalSum of all cards - minimum sum of that window.
 */

class Solution {
    public int maxScore(int[] cardPoints, int k) {

        int n = cardPoints.length; // total number of cards

        int min = 0;      // will store the minimum sum of the "unpicked" middle window
        int totalsum = 0; // sum of all cards
        int sum = 0;       // running sum used while sliding the window

        // Step 1: calculate sum of all cards in the array
        for (int i = 0; i < n; i++) {
            totalsum += cardPoints[i]; // add each card's value to totalsum
        }

        // Step 2: figure out size of the "middle" window we do NOT pick
        // Since we pick k cards total from front+back, the remaining (n-k) 
        // cards in the middle are the ones NOT picked
        int middle = n - k;

        // Step 3: calculate sum of the very first window of size 'middle'
        // (cards from index 0 to middle-1)
        for (int i = 0; i < middle; i++) {
            sum += cardPoints[i]; // add each card in the first window to sum
        }

        // Step 4: set min to this first window's sum (our starting guess)
        min = sum;

        // Step 5: slide the window one step at a time across the array
        // by removing the leftmost card of the previous window
        // and adding the next card on the right
        for (int i = middle; i < n; i++) {

            // remove the card that is leaving the window (i - middle)
            // and add the new card entering the window (i)
            sum = sum - cardPoints[i - middle] + cardPoints[i];

            // keep track of the smallest window sum seen so far
            min = Math.min(sum, min);
        }

        // Step 6: the best score = total sum of all cards - smallest sum of 
        // the middle window we didn't pick (because picking the rest 
        // maximizes what's left after removing the least valuable middle part)
        return totalsum - min;
    }
}
