import java.util.*;

/*
 * ============================================================
 * QUESTION (LeetCode 677 - Map Sum Pairs)
 * ============================================================
 * Design a data structure that supports:
 *   insert(key, val)  -> insert a (string, integer) pair.
 *                        If the key already exists, its value
 *                        is OVERWRITTEN with the new val.
 *   sum(prefix)       -> return the SUM of values of all keys
 *                        that start with the given prefix.
 *
 * Example:
 *   insert("apple", 3)
 *   sum("ap")   -> 3        (only "apple" starts with "ap")
 *   insert("app", 2)
 *   sum("ap")   -> 5        (3 + 2, since both "apple" and "app" start with "ap")
 *
 * ============================================================
 * APPROACH (Trie)
 * ============================================================
 * 1. Build a Trie where each node represents one character.
 * 2. Store the value at the node where a key ENDS
 *    (not on every intermediate node).
 * 3. insert(key, val): walk/create the path for "key",
 *    then set weight = val at the final node.
 *    (Overwrites automatically if key was inserted before.)
 * 4. sum(prefix): walk down to the node representing the
 *    end of "prefix". If that path doesn't exist, return 0.
 *    Otherwise, DFS through EVERY node in that subtree and
 *    add up all the "weight" values found -- this collects
 *    every key that starts with "prefix".
 * ============================================================
 */

class MapSum {

    // A single node in the trie
    private class TrieNode {                 // FIX: renamed "Trie" -> "TrieNode" to match usage below
        TrieNode[] children = new TrieNode[26];
        int weight = 0;                      // value stored ONLY if a key ends exactly here
    }

    private TrieNode root;

    public MapSum() {
        root = new TrieNode();
    }

    public void insert(String key, int val) {
        TrieNode curr = root; // start walking from the root

        for (char c : key.toCharArray()) {
            int idx = c - 'a';

            // create the path if it doesn't exist yet
            if (curr.children[idx] == null) {
                curr.children[idx] = new TrieNode();
            }

            curr = curr.children[idx];  // FIX: actually MOVE curr to the child
        }

        curr.weight = val;  // FIX: use "=" to actually SET the value at the key's end node
                             // (overwrites old value if key was already inserted before)
    }

    public int sum(String prefix) {
        TrieNode curr = root;

        // walk down the trie following the prefix's letters
        for (char c : prefix.toCharArray()) {
            int idx = c - 'a';

            if (curr.children[idx] == null) {
                return 0; // prefix doesn't exist at all -> no keys match
            }

            curr = curr.children[idx];  // FIX: actually MOVE curr to the child
        }

        // curr now sits at the node representing the end of "prefix"
        // sum up every key's weight found anywhere below this node
        return dfs(curr);
    }

    // Recursively adds up weight of this node + weights of everything below it
    private int dfs(TrieNode node) {
        int sum = 0;

        for (int i = 0; i < 26; i++) {
            if (node.children[i] != null) {
                sum += dfs(node.children[i]); // recurse into each existing child
            }
        }

        return sum + node.weight; // add this node's own weight (0 if no key ends here)
    }
}

 * int param_2 = obj.sum(prefix);
