import java.util.*;

/*
 * ============================================================
 * QUESTION (LeetCode 212 - Word Search II)
 * ============================================================
 * Given an m x n board of characters and a list of words,
 * find ALL words from the list that can be formed by
 * tracing a path through ADJACENT cells (up/down/left/right).
 * The same cell cannot be used more than once in a single word.
 *
 * Example:
 * board = [["o","a","a","n"],
 *          ["e","t","a","e"],
 *          ["i","h","k","r"],
 *          ["i","f","l","v"]]
 * words = ["oath","pea","eat","rain"]
 * Output: ["oath","eat"]
 *
 * ============================================================
 * APPROACH (Trie + Backtracking DFS)
 * ============================================================
 * Why not just search the board separately for every word?
 * That would be very slow if there are many words -- we'd
 * redo the same board traversal again and again.
 *
 * Instead:
 * 1. Insert ALL words into a single Trie. Now every word's
 *    prefix path is shared and stored once.
 * 2. Start a DFS from EVERY cell on the board simultaneously
 *    walking down the Trie as we walk the board.
 *    - At each step, only continue if the current board letter
 *      matches an existing child in the Trie (meaning some word
 *      still could be completed this way).
 *    - If we land on a Trie node where isword == true, we've
 *      found a complete valid word -- add it to our results.
 * 3. Use a visited[][] (flag) grid to avoid reusing the same
 *    cell twice in one path, backtracking (un-marking) after
 *    exploring each direction.
 * 4. Use a Set (not a List) to collect results, since the
 *    same word could technically be reachable via multiple
 *    paths on the board -- Set automatically avoids duplicates.
 * ============================================================
 */

class Solution {

    public class TrieNode {
        public boolean isword = false;         // true if a word ends exactly here
        public TrieNode[] child = new TrieNode[26];

        public TrieNode() {
        }
    }

    TrieNode root = new TrieNode();
    boolean[][] flag; // tracks visited cells during current DFS path

    public List<String> findWords(char[][] board, String[] words) {
        Set<String> result = new HashSet<>(); // HashSet avoids duplicate words in output
        flag = new boolean[board.length][board[0].length];

        addToTrie(words); // build the trie from all words first

        // try starting a search from every single cell on the board
        for (int i = 0; i < board.length; i++) {
            for (int j = 0; j < board[0].length; j++) {
                // small optimization: only start searching if this letter
                // could possibly begin ANY word in our trie
                if (root.child[board[i][j] - 'a'] != null) {
                    // FIX: pass "root" as the starting TrieNode (was missing before)
                    search(board, i, j, root, "", result);
                }
            }
        }

        return new LinkedList<>(result); // convert Set to List for the return type
    }

    // Inserts every word into the trie, letter by letter
    private void addToTrie(String[] words) {
        for (String word : words) {
            TrieNode node = root;
            for (int i = 0; i < word.length(); i++) {
                char ch = word.charAt(i);
                if (node.child[ch - 'a'] == null) {
                    node.child[ch - 'a'] = new TrieNode();
                }
                node = node.child[ch - 'a']; // move down into the child
            }
            node.isword = true; // mark end of this word
        }
    }

    /*
     * DFS from cell (i, j), simultaneously walking down the trie.
     * "node" = current trie node matching the path walked so far.
     * "word" = the actual string built so far from board letters.
     */
    private void search(char[][] board, int i, int j, TrieNode node, String word, Set<String> result) {
        // stop conditions:
        // - out of bounds
        // - already visited in this path (avoid reusing a cell)
        // - this letter doesn't continue any word in the trie
        if (i >= board.length || i < 0 || j >= board[i].length || j < 0
                || flag[i][j] || node.child[board[i][j] - 'a'] == null) {
            return;
        }

        flag[i][j] = true; // mark this cell as visited for this path

        node = node.child[board[i][j] - 'a']; // move down the trie to match this letter

        // FIX: was "node.isWord" (capital W) -- must match field name "isword"
        if (node.isword) {
            result.add(word + board[i][j]); // found a complete valid word!
        }

        // explore all 4 directions (up, down, left, right)
        search(board, i - 1, j, node, word + board[i][j], result);
        search(board, i + 1, j, node, word + board[i][j], result);
        search(board, i, j - 1, node, word + board[i][j], result);
        search(board, i, j + 1, node, word + board[i][j], result);

        flag[i][j] = false; // BACKTRACK: unmark so other paths can reuse this cell
    }
}
