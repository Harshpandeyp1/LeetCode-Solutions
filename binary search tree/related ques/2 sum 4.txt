import java.util.HashSet;
import java.util.Set;

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    // Stores the values of nodes visited so far
    Set<Integer> set = new HashSet<>();

    public boolean findTarget(TreeNode root, int k) {
        /*
         * Question: Two Sum IV - Input is a BST (LeetCode 653)
         * - Given the root of a Binary Search Tree and an integer `k`, return true
         *   if there exist two elements in the BST such that their sum equals `k`,
         *   or false otherwise.
         *
         * Approach: Pre-order Traversal with Hash Set (Complement Lookup)
         * - This is the tree variation of the classic "Two Sum" problem using a hash set.
         * - Traverse each node in the tree (e.g., pre-order traversal):
         *     1. Base case: If the current node is null, return false.
         *     2. Complement check: Compute the required counterpart `target = k - root.val`.
         *        If `set` already contains this complement, a valid pair has been found -> return true.
         *     3. Update state: Add `root.val` to the hash set so future nodes can match with it.
         *     4. Recursion: Recursively search the left subtree, then the right subtree.
         *        If either branch finds a valid pair, propagate `true` upwards immediately.
         *
         * Complexity:
         * - Time Complexity: O(N) — each node is visited at most once; set operations (add/contains) are O(1) average.
         * - Space Complexity: O(N) — up to N elements stored in the HashSet, plus O(H) for recursion stack.
         */
        if (root == null) {
            return false;
        }

        // 1. Check if the required complement (k - root.val) was already seen
        if (set.contains(k - root.val)) {
            return true;
        }

        // 2. Add current node's value to the set for upcoming nodes
        set.add(root.val);

        // 3. Search in left subtree; if found, return true immediately
        if (findTarget(root.left, k)) {
            return true;
        }

        // 4. Search in right subtree; if found, return true immediately
        if (findTarget(root.right, k)) {
            return true;
        }

        // No pair found along this path
        return false;
    }
}
