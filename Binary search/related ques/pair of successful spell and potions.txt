/*
 * QUESTION: Successful Pairs of Spells and Potions
 * ---------------------------------------------------
 * You are given two positive integer arrays `spells` and `potions`, of
 * length n and m respectively, where spells[i] is the strength of the
 * ith spell and potions[j] is the strength of the jth potion.
 *
 * You are also given an integer `success`. A spell-potion pair is
 * successful if the PRODUCT of their strengths is >= success.
 *
 * Return an integer array `pairs` of length n where pairs[i] is the
 * number of potions that form a successful pair with the ith spell.
 *
 * Example:
 * spells = [5,1,3], potions = [1,2,3,4,5], success = 7
 * Output: [4,0,3]
 *
 * APPROACH: Sort + Binary Search (lower bound)
 * -----------------------------------------------
 * Brute force (check every spell against every potion) is O(n * m) —
 * too slow for large inputs.
 *
 * Instead:
 * 1. SORT `potions` ascending. This makes products monotonic for a
 *    fixed spell: as potion strength increases, spell*potion increases
 *    too — which is exactly the condition binary search needs.
 *
 * 2. For each spell, binary search on the sorted potions array to find
 *    the LEFTMOST index `idx` such that:
 *          (long) spell * potions[idx] >= success
 *    Since potions is sorted ascending, every potion from `idx` to the
 *    end of the array will ALSO satisfy the condition.
 *
 * 3. So the count of successful potions for this spell is simply:
 *          pairs[i] = m - idx
 *    (m = total potions, idx = first index that qualifies)
 *
 * We use `long` during multiplication to avoid integer overflow, since
 * spell and potion values can be large.
 *
 * Time Complexity:  O((n + m) log m)
 *   - O(m log m) to sort potions once
 *   - O(log m) binary search per spell, done n times
 * Space Complexity: O(1) extra (excluding output array)
 */
class Solution {
    public int[] successfulPairs(int[] spells, int[] potions, long success) {
        int n = spells.length;
        int m = potions.length;

        // Step 1: sort potions ascending so binary search works
        Arrays.sort(potions);

        int[] pairs = new int[n];

        // Step 2 & 3: for each spell, binary search for the boundary
        for (int i = 0; i < n; i++) {
            long spell = spells[i];
            int idx = findLeftmostSuccessful(potions, spell, success);
            pairs[i] = m - idx; // everything from idx onward qualifies
        }

        return pairs;
    }

    /*
     * Binary search for the leftmost index `idx` in sorted `potions`
     * such that (spell * potions[idx]) >= success.
     * If no such index exists, returns potions.length (meaning 0 qualify).
     */
    private int findLeftmostSuccessful(int[] potions, long spell, long success) {
        int left = 0;
        int right = potions.length; // note: length, not length-1
                                     // (right can itself be the "not found" answer)
        int result = potions.length; // default: none qualify

        while (left < right) {
            int mid = left + (right - left) / 2;

            // use long to avoid overflow when multiplying
            long product = spell * (long) potions[mid];

            if (product >= success) {
                result = mid;    // this index qualifies, but an earlier one might too
                right = mid;     // keep searching LEFT for an even earlier qualifying index
            } else {
                left = mid + 1;  // not enough, must look further right
            }
        }

        return result;
    }
}
