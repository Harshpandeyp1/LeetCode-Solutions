/*
 * QUESTION: Search a 2D Matrix II
 * -------------------------------
 * Write an efficient algorithm that searches for a value `target` in an
 * m x n integer matrix. The matrix has these properties:
 *   - Integers in each row are sorted in ascending order (left to right).
 *   - Integers in each column are sorted in ascending order (top to bottom).
 *
 * Note: Row ranges can OVERLAP (e.g. row0 = [1,4,7,11], row1 = [2,5,8,12]),
 * so this is NOT the same as "Search a 2D Matrix I". You cannot flatten it
 * into one sorted array, and you cannot binary-search rows to pick "the
 * one right row" — there's no clean row boundary since ranges overlap.
 *
 * Example:
 * matrix = [[1,4,7,11],[2,5,8,12],[3,6,9,16],[10,13,14,17]], target = 5
 * -> true
 */

class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int m = matrix.length;      // number of rows
        int n = matrix[0].length;   // number of columns

        /*
         * APPROACH: Staircase Search (start from top-right corner)
         * ----------------------------------------------------------
         * Start at row = 0, col = n - 1 (top-right corner).
         *
         * Why top-right? At this cell:
         *   - moving LEFT  -> values decrease (row sorted ascending)
         *   - moving DOWN  -> values increase (column sorted ascending)
         * This gives a clean two-way decision at every step, which is
         * why top-left doesn't work (both directions increase there).
         *
         * At each cell compare matrix[row][col] with target:
         *   - equal   -> found it, return true
         *   - current > target -> everything below in this column is even
         *                         bigger, so this whole column is useless
         *                         -> eliminate the column: col--
         *   - current < target -> everything to the left in this row is
         *                         even smaller, so this whole row is useless
         *                         -> eliminate the row: row++
         *
         * Each step throws away one full row or one full column, so at
         * most (m + n) steps are needed in total.
         *
         * Time Complexity:  O(m + n)
         * Space Complexity: O(1)
         */
        int row = 0;
        int col = n - 1;

        while (row < m && col >= 0) {
            int current = matrix[row][col];

            if (current == target) {
                return true;                 // found the target
            } else if (current > target) {
                col--;                        // eliminate this column
            } else {
                row++;                        // eliminate this row
            }
        }

        // row or col ran out of bounds -> target not present
        return false;
    }
}
