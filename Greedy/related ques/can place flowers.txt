import java.util.*;

/*
========================================================================================
QUESTION:
Given an integer array `flowerbed` containing 0's (empty) and 1's (occupied), and an 
integer `n`, return true if `n` new flowers can be planted without violating the rule 
that no two flowers can be planted in adjacent plots. Otherwise, return false.

APPROACH (Greedy Placement):
1. Iterate through each plot `i` from left to right.
2. For each plot, check if three conditions are satisfied:
   - Current plot is empty (`flowerbed[i] == 0`).
   - Left neighbor is empty OR current plot is at the left boundary (`i == 0`).
   - Right neighbor is empty OR current plot is at the right boundary (`i == len - 1`).
3. If all conditions are met, greedily plant a flower (`flowerbed[i] = 1`) and decrement `n`.
4. Early loop exit occurs when `n` drops to 0. Return `true` if `n <= 0`.

TIME COMPLEXITY: O(N) — Single pass through the flowerbed array.
SPACE COMPLEXITY: O(1) — Modifies array in-place with no extra memory allocation.
========================================================================================
*/

class Solution {
    public boolean canPlaceFlowers(int[] flowerbed, int n) {
        // Cache total length of the flowerbed array
        int len = flowerbed.length;

        // Iterate through each plot, stopping early if all needed flowers are planted (n <= 0)
        for (int i = 0; i < len && n > 0; i++) {
            
            // Left plot is valid if current plot is at boundary index 0 OR left neighbor is empty
            boolean leftEmpty  = (i == 0)       || flowerbed[i - 1] == 0;
            
            // Right plot is valid if current plot is at boundary index len-1 OR right neighbor is empty
            boolean rightEmpty = (i == len - 1) || flowerbed[i + 1] == 0;

            // Check if current plot is empty and both left & right neighbors are empty
            if (flowerbed[i] == 0 && leftEmpty && rightEmpty) {
                
                // Plant a flower at current position
                flowerbed[i] = 1;   
                
                // Decrement count of remaining flowers to plant
                n--;
            }
        }
        
        // Return true if we successfully planted at least 'n' flowers
        return n <= 0;
    }
}
