class Solution {
    /*
     * Question:
     * There are n children standing in a line. Each child has a rating.
     * Distribute candies such that:
     * 1. Every child gets at least one candy.
     * 2. A child with a higher rating than an adjacent child must get more candies.
     * Return the minimum number of candies required.
     *
     * Approach (Two-Pass Greedy):
     * 1. Initialize every child with 1 candy.
     * 2. Traverse from left to right:
     *    - If the current child's rating is greater than the left child's rating,
     *      give one more candy than the left child.
     * 3. Traverse from right to left:
     *    - If the current child's rating is greater than the right child's rating,
     *      ensure it has one more candy than the right child.
     *    - Use Math.max() so that the left-to-right assignment is not overwritten.
     * 4. Sum all candies and return the result.
     *
     * Time Complexity: O(n)
     * Space Complexity: O(n)
     */

    public int candy(int[] ratings) {

        int n = ratings.length;

        // Step 1: Give every child at least one candy.
        int[] candies = new int[n];
        Arrays.fill(candies, 1);

        // Step 2: Left to Right
        // If current rating is greater than previous,
        // assign one more candy than previous child.
        for (int i = 1; i < n; i++) {
            if (ratings[i] > ratings[i - 1]) {
                candies[i] = candies[i - 1] + 1;
            }
        }

        // Step 3: Right to Left
        // If current rating is greater than next,
        // ensure current child has one more candy than next child.
        for (int i = n - 2; i >= 0; i--) {
            if (ratings[i] > ratings[i + 1]) {
                candies[i] = Math.max(candies[i], candies[i + 1] + 1);
            }
        }

        // Step 4: Calculate total candies.
        int total = 0;
        for (int candy : candies) {
            total += candy;
        }

        return total;
    }
}
