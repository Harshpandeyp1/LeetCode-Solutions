/**
 * LeetCode 70 - Climbing Stairs
 *
 * Question:
 * You are climbing a staircase. It takes n steps to reach the top.
 * Each time you can either climb 1 or 2 steps.
 * In how many distinct ways can you climb to the top?
 *
 * Example:
 * n = 2 -> 2 ways (1+1, 2)
 * n = 3 -> 3 ways (1+1+1, 1+2, 2+1)
 *
 * Approach: Dynamic Programming (Bottom-Up / Tabulation)
 * -------------------------------------------------------
 * This is essentially the Fibonacci sequence in disguise:
 *   - To reach step i, you can either come from step (i-1) by taking
 *     a single 1-step, OR from step (i-2) by taking a 2-step.
 *   - So: ways(i) = ways(i-1) + ways(i-2)
 *
 * Base cases:
 *   - dp[1] = 1  -> only one way to climb 1 step (1)
 *   - dp[2] = 2  -> two ways to climb 2 steps (1+1, 2)
 *
 * Build up dp[3..n] using the recurrence above and return dp[n].
 *
 * Time Complexity:  O(n)  -> single loop from 3 to n
 * Space Complexity: O(n)  -> dp array of size n+1
 *                    (can be optimized to O(1) using two variables
 *                    instead of a full array, since we only ever
 *                    need the previous two values)
 */
class Solution {
    public int climbStairs(int n) {
        // Edge case: only 1 step means only 1 way to climb
        if (n == 1) return 1;

        // dp[i] = number of distinct ways to reach step i
        int dp[] = new int[n + 1];

        dp[1] = 1; // base case: 1 way to reach step 1
        dp[2] = 2; // base case: 2 ways to reach step 2

        // Fill dp[3..n] using recurrence: dp[i] = dp[i-1] + dp[i-2]
        for (int i = 3; i <= n; i++) {
            dp[i] = dp[i - 1] + dp[i - 2];
        }

        // dp[n] holds the total number of distinct ways to reach the top
        return dp[n];
    }
}
