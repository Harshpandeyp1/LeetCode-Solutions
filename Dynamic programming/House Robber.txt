/**
 * LeetCode 198 - House Robber
 *
 * Question:
 * You are a robber planning to rob houses along a street. Each house has
 * a certain amount of money stashed, given in the array `nums`.
 * You cannot rob two adjacent houses (robbing both triggers the alarm).
 * Find the maximum amount of money you can rob without alerting the police.
 *
 * Example:
 * nums = [1,2,3,1] -> 4 (rob house 0 and house 2: 1+3=4)
 * nums = [2,7,9,3,1] -> 12 (rob house 0, 2, 4: 2+9+1=12)
 *
 * Approach: Dynamic Programming (Bottom-Up / Tabulation)
 * -------------------------------------------------------
 * For every house i, you have two choices:
 *   1. Skip house i -> best you can do is dp[i-1] (whatever max was up to previous house)
 *   2. Rob house i  -> nums[i] + dp[i-2] (can't rob adjacent house i-1, so add to
 *                       the best result up to two houses back)
 *
 * So the recurrence is:
 *   dp[i] = max(dp[i-1], dp[i-2] + nums[i])
 *
 * Base cases:
 *   - dp[0] = nums[0]                  -> only one house, rob it
 *   - dp[1] = max(nums[0], nums[1])    -> pick the richer of the first two houses
 *
 * dp[i] represents the maximum money that can be robbed considering
 * houses 0..i. Final answer is dp[nums.length - 1].
 *
 * Time Complexity:  O(n)  -> single pass through the array
 * Space Complexity: O(n)  -> dp array of size n
 *                    (can be optimized to O(1) using two variables,
 *                    since dp[i] only depends on dp[i-1] and dp[i-2])
 */
class Solution {
    public int rob(int[] nums) {
        // Edge case: 0 or 1 house -> just return that house's value
        // (nums.length == 0 would actually throw here; add a guard if needed)
        if (nums.length < 2) return nums[0];

        // dp[i] = max money robbable considering houses 0..i
        int dp[] = new int[nums.length];

        dp[0] = nums[0];                          // base case: only house 0 available
        dp[1] = Math.max(nums[0], nums[1]);        // base case: best of first two houses

        // Fill dp[2..n-1] using recurrence: dp[i] = max(dp[i-1], dp[i-2] + nums[i])
        for (int i = 2; i < nums.length; i++) {
            dp[i] = Math.max(dp[i - 2] + nums[i], dp[i - 1]);
        }

        // dp[n-1] holds the max money robbable from the entire street
        return dp[nums.length - 1];
    }
}
