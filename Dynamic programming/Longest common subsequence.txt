/**
 * QUESTION: Longest Common Subsequence
 * ------------------------------------
 * Given two strings text1 and text2, find the length of their longest
 * common subsequence (LCS). A subsequence is a sequence that appears
 * in the same relative order, but not necessarily contiguously.
 * If there is no common subsequence, return 0.
 *
 * Example:
 *   text1 = "ace", text2 = "abcde"
 *   LCS = "ace" -> length 3
 *
 * APPROACH: 2D Dynamic Programming
 * ---------------------------------
 * Build a table dp[i][j] = length of LCS using the first i characters
 * of text1 and the first j characters of text2.
 *
 * - If characters text1[i-1] and text2[j-1] match:
 *     this letter is part of the LCS, so extend the diagonal answer:
 *     dp[i][j] = dp[i-1][j-1] + 1
 *
 * - If they don't match:
 *     we can't use both letters together here, so try dropping one
 *     letter from either string and keep whichever gives a longer LCS:
 *     dp[i][j] = max(dp[i-1][j], dp[i][j-1])
 *
 * Final answer = dp[n][m], using the FULL length of both strings.
 *
 * Time Complexity:  O(n * m)  -> we fill an (n+1) x (m+1) grid
 * Space Complexity: O(n * m)  -> for the dp table
 */
class Solution {
    public int longestCommonSubsequence(String text1, String text2) {

        int n = text1.length();               // length of first string
        int m = text2.length();                // length of second string

        // dp[i][j] = LCS length using first i chars of text1, first j chars of text2
        // extra row/col of 0s (index 0) represents "using 0 characters" -> LCS = 0
        int[][] dp = new int[n + 1][m + 1];

        // i represents how many characters of text1 we've considered so far (1 to n)
        for (int i = 1; i <= n; i++) {

            // j represents how many characters of text2 we've considered so far (1 to m)
            for (int j = 1; j <= m; j++) {

                // compare the CURRENT characters: text1[i-1] and text2[j-1]
                // (we use i-1, j-1 because dp is 1-indexed but strings are 0-indexed)
                if (text1.charAt(i - 1) == text2.charAt(j - 1)) {

                    // characters match! this letter belongs in the common subsequence.
                    // take the best answer WITHOUT this letter on either side (diagonal)
                    // and add 1 to account for this newly matched letter
                    dp[i][j] = dp[i - 1][j - 1] + 1;

                } else {

                    // characters are different, so this exact pair can't both
                    // be part of the LCS at the same time.
                    //
                    // two choices:
                    // 1. drop the current text1 character -> look at dp[i-1][j]
                    // 2. drop the current text2 character -> look at dp[i][j-1]
                    //
                    // take whichever option gives the longer LCS so far
                    dp[i][j] = Math.max(
                        dp[i - 1][j],   // ignore text1's current character
                        dp[i][j - 1]    // ignore text2's current character
                    );
                }
            }
        }

        // dp[n][m] = LCS using ALL of text1 (n chars) and ALL of text2 (m chars)
        // this is the final answer
        return dp[n][m];
    }
}
