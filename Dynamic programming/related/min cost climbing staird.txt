class Solution {
    /*
        Problem: Min Cost Climbing Stairs
        --------------------------------
        cost[i] = cost of stepping on stair i.
        From any stair, you can climb 1 or 2 steps.
        You may start from index 0 or index 1.
        Find the minimum cost to reach the "top" 
        (i.e., one step beyond the last index, index n).

        Approach: Bottom-Up Dynamic Programming
        ----------------------------------------
        dp[i] = minimum cost to REACH stair i (before paying to leave it)

        Base cases:
            dp[0] = 0  -> starting point, no cost to "reach" it
            dp[1] = 0  -> also a valid starting point

        Transition:
            dp[i] = min( dp[i-1] + cost[i-1],   // come from previous stair
                         dp[i-2] + cost[i-2] )  // come from two stairs before

        Answer:
            dp[n] -> cost to reach just past the last stair (the top)

        Time Complexity  : O(n)
        Space Complexity : O(n)  (can be reduced to O(1) with two variables)
    */
    public int minCostClimbingStairs(int[] cost) {
        int n = cost.length;

        // dp[i] represents min cost to reach stair i
        int dp[] = new int[n + 1];

        // dp[0] = 0 and dp[1] = 0 by default (Java initializes int arrays to 0)
        // so no explicit assignment needed here

        // Fill dp array from stair 2 up to stair n (the "top")
        for (int i = 2; i <= n; i++) {
            // Option 1: come from stair (i-1), paying cost[i-1]
            int fromOneStepBack = cost[i - 1] + dp[i - 1];

            // Option 2: come from stair (i-2), paying cost[i-2]
            int fromTwoStepsBack = cost[i - 2] + dp[i - 2];

            // Take the cheaper option
            dp[i] = Math.min(fromOneStepBack, fromTwoStepsBack);
        }

        // dp[n] holds the minimum cost to reach the top of the staircase
        return dp[n];
    }
}
