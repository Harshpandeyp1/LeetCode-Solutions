/*
 * QUESTION:
 * There is one store owner who is running a store. On a given day with n 
 * minutes, every minute has some number of customers coming in, given by 
 * an array `customers`. Every minute, the store owner is either grumpy or 
 * not, given by a binary array `grumpy` (1 = grumpy that minute, 0 = not 
 * grumpy). When the owner is grumpy, the customers of that minute are NOT 
 * satisfied. Otherwise, they are satisfied.
 * The owner knows a secret technique to keep themselves not grumpy for 
 * `minutes` consecutive minutes, but can only use it once.
 * Return the maximum number of customers that can be satisfied throughout 
 * the day.
 *
 * APPROACH:
 * - Customers during minutes when the owner is already NOT grumpy are 
 *   satisfied no matter what — count these first as a fixed base total 
 *   (stored in maxsum).
 * - Customers during minutes when the owner IS grumpy are only satisfied 
 *   if that minute falls inside the special window we activate.
 * - So we need to find the window of size `minutes` that captures the 
 *   maximum number of "grumpy" customers — these are the extra customers 
 *   we can "save" by using the technique. This is tracked using sum/max.
 * - Use a sliding window of size `minutes` to find this maximum extra sum.
 * - Final answer = maxsum (base satisfied) + max (best extra saved).
 */

class Solution {
    public int maxSatisfied(int[] customers, int[] grumpy, int minutes) {

        int n = customers.length; // total number of minutes in the day
        int m = grumpy.length;     // total number of minutes in grumpy array (same as n)

        int max = 0;     // will store the maximum extra customers saved by the window
        int sum = 0;      // running sum of grumpy-minute customers inside current window
        int maxsum = 0;   // base satisfied customers (owner not grumpy) — calculated once, stays fixed

        // Step 1: calculate base satisfied customers
        // (only count minutes where owner is NOT grumpy)
        for (int i = 0; i < n; i++) {
            if (grumpy[i] == 0) {
                maxsum += customers[i]; // these customers are always satisfied, add to base
            }
        }
        // NOTE: maxsum is now fixed and represents the base satisfied count,
        // it will NOT be touched again below

        // Step 2: figure out size of the sliding window (the technique's duration)
        int maxwindow = minutes;

        // Step 3: build the very first window of size 'maxwindow'
        // sum only the customers where owner IS grumpy (i.e., customers 
        // we could "save" by activating the technique here)
        for (int i = 0; i < maxwindow; i++) {
            if (grumpy[i] == 1) {
                sum += customers[i]; // add grumpy-minute customers to window sum
            }
        }

        // Step 4: set max to this first window's sum (starting guess)
        max = sum;

        // Step 5: slide the window one minute at a time across the rest of the day
        for (int i = maxwindow; i < n; i++) {

            // add the new customer entering the window, but only if 
            // that minute is a grumpy minute (otherwise it's already 
            // counted in maxsum, so adding it would double count)
            if (grumpy[i] == 1) {
                sum += customers[i];
            }

            // remove the customer leaving the window (the one at i - maxwindow),
            // but only if that minute was a grumpy minute
            if (grumpy[i - maxwindow] == 1) {
                sum -= customers[i - maxwindow];
            }

            // keep track of the largest extra sum seen so far
            max = Math.max(sum, max);
        }

        // Step 6: final answer = base satisfied customers (maxsum) + best extra saved (max)
        return maxsum + max;
    }
}
