
Given a string s, find the length of the longest substring without duplicate characters.


class Solution {
    public int lengthOfLongestSubstring(String s) {
     int n=s.length();
     int left=0;
     int right=0;
     int ans=0;
     int freq[]=new int[128];

     for(right=0;right<n;right++){//moving right pointer
        char ch=s.charAt(right);//if occur inc freq
        freq[ch]++;
     
     while(freq[ch]>1){//if nsy freq is greater then one remove left elm and inc left pointer
        freq[s.charAt(left)]--;
        left++;
     }
     ans = Math.max(ans, right - left + 1);//max of ans of last indx of substring right-left+1 formula
    }
    return ans;
    }

}
