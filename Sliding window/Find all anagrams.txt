Given two strings s and p, return an array of all the start indices of p's anagrams in s. You may return the answer in any order.

 

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        int n=s.length();
        int m=p.length();
        List<Integer>list=new ArrayList<>();//ans arraylist
        if(m>n) return list;
         int Scount[]=new int [26];
         int Pcount[]=new int[26];
        for(int i=0;i<m;i++){
            int si=s.charAt(i)-'a';//couting freq through ascii value  a-a=0 b-a=1 z-a=25 of the first  winfow
            int pi=p.charAt(i)-'a';
            Scount[si]++;
            Pcount[pi]++;
            
        }
        if(Arrays.equals(Scount,Pcount)){//compare freq of the first winddow with the pattern
            list.add(0);
        }
        for(int i=1;i<=n-m;i++){ 
            int prev=s.charAt(i-1)-'a';//reduce prev elm freq and take one next elm in windpw
            int next=s.charAt(i+m-1)-'a';//taking the next elm
            Scount[prev]--;//reducing the frq of prev 
            Scount[next]++;//inc of next
            if(Arrays.equals(Scount,Pcount)){//compare and add
                list.add(i);
            }
        }
        return list;
    }
}
