class Solution {
    public boolean isIsomorphic(String s, String t) {
        HashMap<Character,Character>map=new HashMap<>();
        if (s.length() != t.length()) return false;
        for(int i=0;i<s.length();i++){
            char ss=s.charAt(i);
            char tt=t.charAt(i);
            if(map.containsKey(ss)&&map.get(ss)!=tt||(!map.containsKey(ss)&&map.values().contains(tt))){
//                 //map.containsKey(ss) → "I've already assigned ss a partner before."
// map.get(ss) != tt → "But right now, ss is paired with a different character than last time."!map.containsKey(ss) → "ss is new, I haven't assigned it anything yet."
// map.values().contains(tt) → "But tt is already being used as someone else's partner."
                return false;
            }
             map.put(ss,tt);//add both into the map

             
        }
       

    
    return true;
    }
}
