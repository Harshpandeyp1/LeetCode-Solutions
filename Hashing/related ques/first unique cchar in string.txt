/*
 * LEETCODE 387 - FIRST UNIQUE CHARACTER IN A STRING
 * ----------------------------------------------------
 * QUESTION:
 * Given a string s, find the first character in it that does NOT
 * repeat anywhere else in the string, and return its index.
 * If no such character exists, return -1.
 *
 * Example: s = "leetcode"     -> 'l' is first unique -> return 0
 *          s = "loveleetcode" -> 'v' is first unique -> return 2
 *          s = "aabb"         -> no unique character  -> return -1
 *
 * APPROACH:
 * 1. FIRST PASS: build a frequency map (character -> count) by
 *    scanning the whole string once and counting occurrences of
 *    each character.
 * 2. SECOND PASS: scan the string again from left to right, and for
 *    each character check its count in the map. The FIRST character
 *    whose count equals 1 is the answer, since we're scanning in
 *    original order this guarantees it's the first unique one.
 * 3. If no character has a count of 1, return -1.
 *
 * WHY TWO PASSES INSTEAD OF ONE:
 * We can't know if a character is "unique" while looking at it just
 * once, since a duplicate of it might appear LATER in the string.
 * So we must first know the full count of every character (pass 1)
 * before we can safely check counts in order (pass 2).
 *
 * COMPLEXITY:
 *   Time:  O(n) - two linear passes over the string
 *   Space: O(1) - at most 26 lowercase letters possible (bounded alphabet),
 *                 so the map size doesn't grow with input size
 */

class Solution {
    public int firstUniqChar(String s) {
        // count frequency of each character across the whole string
        Map<Character, Integer> freq = new HashMap<>();
        for (char c : s.toCharArray()) {
            freq.put(c, freq.getOrDefault(c, 0) + 1);
        }

        // second pass: find the first character whose count is exactly 1
        for (int i = 0; i < s.length(); i++) {
            if (freq.get(s.charAt(i)) == 1) {
                return i; // first unique character found
            }
        }

        return -1; // no unique character exists
    }
}
