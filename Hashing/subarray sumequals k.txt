class Solution {
    public int subarraySum(int[] nums, int k) {
        int res = 0; // stores the final count of subarrays that sum to k
        int curr = 0; // running prefix sum as we scan through the array

        Map<Integer, Integer> map = new HashMap<>(); // maps: prefixSum value -> how many times it has occurred so far
        map.put(0, 1); // seed with prefixSum 0 occurring once, to handle subarrays starting from index 0

        for (int i : nums) { // iterate through each element in nums
            curr += i; // update running prefix sum by adding the current element

            res += map.getOrDefault(curr - k, 0); 
            // check how many times (curr - k) has occurred as a prefix sum before;
            // each occurrence marks a valid subarray ending at the current index that sums to k

            map.put(curr, map.getOrDefault(curr, 0) + 1); 
            // record the current prefix sum in the map, incrementing its frequency by 1
        }

        return res; // return the total count of subarrays summing to k
    }
